== Install

[source,bash]
----
ansible-playbook mariadb/podman/ansible/podman-install-playbook.yaml \
  -e pod_name=${POD_NAME:-mariadb} \
  -e db_name=${DB_NAME:-mariadb} \
  -e db_user=${DB_USER:-mariadb} \
  -e db_password=${DB_PASSWORD:-mariadb}
----


Template the helm file.

[source,bash]
----
jinja2 --format=yaml -DDB_PASSWORD=${DB_PASSWORD} -DAPP_NAME=${APP_NAME} mariadb/k8s/helm/helm-mariadb-values.yaml.j2 > /tmp/helm-mariadb-values.yaml
----

[source,bash]
----
helm upgrade --install --namespace ${APP_NAME} -f /tmp/helm-odoo-odoo-values.yaml odoo ${HELM_CHART_FOLDER:-.}/MariaDB
----

[source,bash]
----
ansible-playbook mariadb/k8s/helm/ansible/mariadb-install-playbook.yaml \
  -e app_name=${APP_NAME} \
  -e app_version=${APP_VERSION} \
  -e trikora_helm_project_dir=${TRIKORA_HELM} \
  -e app_namespace=${APP_NAME} \
  -e db_password=${DB_PASSWORD} \
  -e db_name=${DB_PASSWORD} \
  -e db_user=${APP_NAME} \
  -e db_image_tag=13

  -e k8s_host=${KUBERNETES_NODE_NAME} \

----

Use the application variables file.

[source,bash]
----
ansible-playbook mariadb/k8s/helm/ansible/mariadb-install-playbook.yaml \
  -e "@gitea/k8s/gitea_variables.yaml" \
  -e trikora_helm_project_dir=${TRIKORA_HELM} \
  -e db_password=${DB_PASSWORD} \
  -e db_image_tag=13 \
  -e iscsi_portal=${ISCSI_PORTAL} \
  -e iscsi_target=${ISCSI_TARGET}
----

== Uninstall

[source,bash]
----
ansible-playbook mariadb/k8s/helm/ansible/mariadb-uninstall-playbook.yaml -e "@gitea/k8s/gitea_variables.yaml"   -e trikora_helm_project_dir=${TRIKORA_HELM}
----
